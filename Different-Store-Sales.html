ALTER  TABLE Different_stores_dataset

In [24]:
cursor = conn.cursor()
cursor.execute("""
UPDATE Different_stores_dataset
SET
    Day_Name = DATENAME(WEEKDAY, Invoice_Date),
    Month_Name = DATENAME(MONTH, Invoice_Date),
    Profit = (Selling_Price_Per_Unit - Cost_Price_Per_Unit) * Quantity,
    Profit_Margin = 
        ((Selling_Price_Per_Unit - Cost_Price_Per_Unit)
        / NULLIF(Selling_Price_Per_Unit, 0)) * 100
""")

conn.commit()
